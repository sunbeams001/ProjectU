## 🗺️ 导航架构

### 导航层级结构

```
App (Navigator)
├── LoginScreen (未登录时)
│   └── 登录成功 → replaceAll(HomeScreen)
│
└── HomeScreen (已登录时，TabNavigator)
    ├── HomeTab (主页)
    ├── DiscoveryTab (发现)
    │   ├── → DiscoveryUsersScreen (推荐用户)
    │   ├── → DiscoveryIllustsScreen (推荐插画)
    │   │   └── → ArtworkDetailScreen (作品详情，支持列表导航+缓存+返回定位)
    │   └── → DiscoveryNovelsScreen (推荐小说)
    ├── RankingTab (排行榜，支持多模式+返回定位)
    │   └── → ArtworkDetailScreen (作品详情，支持列表导航+缓存+返回定位)
    └── ProfileTab (个人)
        └── → SettingsScreen (设置)
            └── → ApiTestScreen (API测试)
```

### 导航实现方式

**使用 Voyager 导航框架**：
- **Navigator**: 全局页面栈导航
- **TabNavigator**: Tab 页签导航（主页底部导航）
- **Screen**: 所有页面继承自 `Screen` 接口

**导航操作**：
```kotlin
// 跳转到新页面
navigator.push(SettingsScreen())

// 替换当前页面
navigator.replace(HomeScreen())

// 替换所有页面（登录成功）
navigator.replaceAll(HomeScreen())

// 返回上一页
navigator.pop()

// Tab 切换
tabNavigator.current = DiscoveryTab
```

```kotlin
private fun preloadAdjacentArtworks(currentIndex: Int) {

    if (currentIndex > 0 && !artworkCache.containsKey(artworkIds[currentIndex - 1])) {
        loadArtworkDetail(artworkIds[currentIndex - 1], silent = true)
    }
    if (currentIndex < artworkIds.size - 1 && !artworkCache.containsKey(artworkIds[currentIndex + 1])) {
        loadArtworkDetail(artworkIds[currentIndex + 1], silent = true)
    }
}
```

**RankingScreen.kt - 响应式滚动**:
```kotlin
HorizontalPager(...) { page ->
    val mode = supportedModes[page]
    val listState = rememberLazyStaggeredGridState()
    
    // 使用 derivedStateOf 建立响应式依赖
    val targetScrollIndex by remember(mode.value) {
        derivedStateOf { scrollIndices[mode.value] }
    }
    
    // 监听变化并自动滚动
    LaunchedEffect(targetScrollIndex) {
        val scrollIndex = targetScrollIndex
        if (scrollIndex != null && scrollIndex > 0) {
            listState.scrollToItem(scrollIndex)
            scrollIndices.remove(mode.value)  // 清除标记
        }
    }
    
    ArtworkStaggeredGridLayout(listState = listState, ...)
}
```

**关键技术点**:
- `scrollIndices` 提升到 Tab 对象级别，避免导航时丢失
- `derivedStateOf` 监听 Map 内容变化
- `LaunchedEffect` 自动执行滚动操作
- 滚动后清除标记，避免重复滚动

**返回定位实现**:

**HomeScreen.kt - Tab 级别状态管理**:
```kotlin
object RankingTab : Tab {

    private val scrollIndices = mutableStateMapOf<String, Int>()
    
    @Composable
    override fun Content() {
        RankingContent(
            scrollIndices = scrollIndices,
            onArtworkClick = { artwork, index ->

                val artworkIds = currentModeData?.artworks?.map { it.id } ?: emptyList()
                parentNavigator?.push(
                    ArtworkDetailScreen(
                        artworkIds = artworkIds,
                        initialIndex = index,
                        onLoadMore = { viewModel.loadMore() },
                        onReturnWithIndex = { lastIndex ->
                            // 保存到对应模式
                            scrollIndices[state.currentMode.value] = lastIndex
                        }
                    )
                )
            }
        )
    }
}
```

**RankingScreen.kt - 响应式滚动**:
```kotlin
HorizontalPager(...) { page ->
    val mode = supportedModes[page]
    
    // 为每个模式创建独立的列表状态
    val listState = if (state.currentContentType == RankingContent.NOVEL) {
        rememberLazyListState()
    } else {
        rememberLazyStaggeredGridState()
    }
    
    // 使用 derivedStateOf 建立响应式依赖
    val targetScrollIndex by remember(mode.value) {
        derivedStateOf { scrollIndices[mode.value] }
    }
    
    // 监听变化并自动滚动
    LaunchedEffect(targetScrollIndex) {
        val scrollIndex = targetScrollIndex
        if (scrollIndex != null && scrollIndex > 0) {
            when (listState) {
                is LazyStaggeredGridState -> listState.scrollToItem(scrollIndex)
                is LazyListState -> listState.scrollToItem(scrollIndex)
            }
            scrollIndices.remove(mode.value)  // 清除标记
        }
    }
    
    // 渲染内容
    when (state.currentContentType) {
        RankingContent.NOVEL -> NovelListLayout(...)
        else -> ArtworkStaggeredGridLayout(...)
    }
}
```

**关键技术点**:
1. **两层 Pager 结构**：
   - 第一层：内容类型切换（通过按钮）
   - 第二层：模式切换（通过 HorizontalPager）

2. **状态缓存到 Map**：
   - listStates: 缓存每个模式的滚动状态
   - modeDataCache: 缓存每个模式的数据
   - scrollIndices: 缓存每个模式的返回滚动位置

3. **响应式滚动**：
   - 使用 `derivedStateOf` 监听 Map 内容变化
   - `LaunchedEffect` 自动执行滚动操作
   - 滚动完成后清除标记，避免重复

4. **惰性加载**：
   - 只加载当前显示的模式数据
   - 切换到新模式时才加载数据
   - 已加载的模式保持数据缓存

## 🎭 MVI 架构模式

### 标准 MVI 实现

所有页面都遵循统一的 MVI 架构模式：

```kotlin
// 1. State - 页面状态
data class XxxScreenState(
    val data: List<Item> = emptyList(),
    val isLoading: Boolean = false,
    val error: String? = null
)

// 2. Intent - 用户操作
sealed interface XxxIntent {
    data object LoadData : XxxIntent
    data class OnItemClick(val id: String) : XxxIntent
}

// 3. ViewModel - 状态管理
class XxxViewModel(
    private val repository: Repository
) : ScreenModel {
    private val _state = MutableStateFlow(XxxScreenState())
    val state: StateFlow<XxxScreenState> = _state.asStateFlow()
    
    init {
        loadData()
    }
    
    fun handleIntent(intent: XxxIntent) {
        when (intent) {
            is XxxIntent.LoadData -> loadData()
            is XxxIntent.OnItemClick -> handleClick(intent.id)
        }
    }
    
    private fun loadData() {
        screenModelScope.launch {
            _state.update { it.copy(isLoading = true) }
            repository.getData()
                .onSuccess { data -> 
                    _state.update { it.copy(data = data, isLoading = false) }
                }
                .onFailure { error ->
                    _state.update { it.copy(error = error.message, isLoading = false) }
                }
        }
    }
}

// 4. Screen - UI 实现
class XxxScreen : Screen {
    @Composable
    override fun Content() {
        val viewModel = koinScreenModel<XxxViewModel>()
        val state by viewModel.state.collectAsState()
        
        XxxScreenContent(
            state = state,
            onIntent = viewModel::handleIntent
        )
    }
}

@Composable
private fun XxxScreenContent(
    state: XxxScreenState,
    onIntent: (XxxIntent) -> Unit
) {
    // UI 实现
}
```

## 🔄 列表导航与返回定位最佳实践

### 核心概念

**列表导航**：从列表页面点击进入详情页后，支持左右滑动浏览列表中的其他项目。

**返回定位**：从详情页返回列表时，列表自动滚动到最后浏览的项目位置。

### 实现要素

#### 1. 状态提升到正确的级别

**问题**：使用 `remember` 在 Composable 中保存状态，导航后会丢失。

**解决方案**：将需要在导航间保持的状态提升到 Tab 对象级别。

```kotlin
object RankingTab : Tab {
    // ✅ 正确：提升到 Tab 对象级别
    private val scrollIndices = mutableStateMapOf<String, Int>()
    
    @Composable
    override fun Content() {
        // ❌ 错误：在 Composable 中使用 remember
        // val scrollIndices = remember { mutableStateMapOf<String, Int>() }
        
        RankingContent(scrollIndices = scrollIndices, ...)
    }
}
```

#### 2. 建立响应式依赖

**问题**：Map 内容变化不会触发 Compose 重组。

**解决方案**：使用 `derivedStateOf` 建立响应式依赖。

```kotlin
// ❌ 错误：直接读取 Map
val targetScrollIndex = scrollIndices[mode.value]

// ✅ 正确：使用 derivedStateOf
val targetScrollIndex by remember(mode.value) {
    derivedStateOf { scrollIndices[mode.value] }
}

LaunchedEffect(targetScrollIndex) {
    // 当 targetScrollIndex 变化时，此处会重新执行
    if (targetScrollIndex != null && targetScrollIndex > 0) {
        listState.scrollToItem(targetScrollIndex)
        scrollIndices.remove(mode.value)
    }
}
```

#### 3. 拦截系统返回

**问题**：用户使用系统返回键或手势返回时，onBackClick 不会被调用。

**解决方案**：使用 `BackHandler` 统一处理所有返回操作。

```kotlin
val handleBack: () -> Unit = {
    val currentIndex = viewModel.getCurrentIndex()
    onReturnWithIndex?.invoke(currentIndex)
    navigator.pop()
}

// 拦截系统返回
BackHandler(enabled = true, onBack = handleBack)

// UI 返回按钮也使用同一处理
IconButton(onClick = handleBack) {
    Icon(Icons.AutoMirrored.Filled.ArrowBack, null)
}
```

#### 4. 缓存机制

**目的**：已浏览的内容瞬间显示，提升用户体验。

**实现**：
```kotlin
// ViewModel 中
private val artworkCache = mutableMapOf<String, Artwork>()

fun loadArtworkDetail(artworkId: String) {
    // 检查缓存
    artworkCache[artworkId]?.let { cached ->
        _state.update { it.copy(artwork = cached) }
        return
    }
    
    // 加载并缓存
    viewModelScope.launch {
        val result = repository.getArtworkDetail(artworkId)
        result.onSuccess { artwork ->
            artworkCache[artworkId] = artwork
            _state.update { it.copy(
                artwork = artwork,
                artworkCache = artworkCache.toMap()
            )}
        }
    }
}
```

#### 5. 预加载策略

**目的**：用户滑动到相邻内容时，内容已加载完成。

**实现**：
```kotlin
private fun preloadAdjacentArtworks(currentIndex: Int) {
    // 预加载前一个
    if (currentIndex > 0) {
        val prevId = artworkIds[currentIndex - 1]
        if (!artworkCache.containsKey(prevId)) {
            loadArtworkDetail(prevId, silent = true)  // 静默加载
        }
    }
    
    // 预加载后一个
    if (currentIndex < artworkIds.size - 1) {
        val nextId = artworkIds[currentIndex + 1]
        if (!artworkCache.containsKey(nextId)) {
            loadArtworkDetail(nextId, silent = true)
        }
    }
}
```

#### 6. ID 匹配检查

**目的**：防止显示旧内容造成混淆。

**实现**：
```kotlin
when {
    cachedArtwork != null -> {
        // 有缓存，瞬间显示
        ArtworkDetailLayout(artwork = cachedArtwork, ...)
    }
    
    pageIndex == state.currentIndex && 
    state.artwork != null && 
    state.artwork.id == artworkId -> {  // ⭐ ID 匹配检查
        // 当前页面且作品ID匹配
        ArtworkDetailLayout(artwork = state.artwork, ...)
    }
    
    else -> {
        // 显示加载指示器
        CircularProgressIndicator()
    }
}
```

### 常见陷阱

| 问题 | 原因 | 解决方案 |
|-----|------|---------|
| 返回后状态丢失 | 状态保存在 Composable 的 remember 中 | 提升到 Tab 对象级别 |
| Map 变化不触发重组 | 直接读取 Map 值 | 使用 derivedStateOf |
| 系统返回不触发回调 | 只处理了 onClick | 使用 BackHandler |
