# 搜索系统设计指南

> 📅 创建日期: 2025-12-10  
> 📝 状态: 设计中  
> 🎯 目标: 实现统一的多Tab搜索界面，支持插画、小说、用户搜索

## 📖 概述

搜索系统是 ProjectU 的核心功能之一，提供对插画/漫画/动图、小说和用户的统一搜索入口。搜索流程分为两个主要页面：**搜索准备页面**和**搜索结果页面**。

### 核心特性

- ✅ **双页面设计**: 搜索准备页 + 搜索结果页
- ✅ **智能建议**: 搜索历史 + 推荐标签 + 实时自动补全
- ✅ **统一筛选**: Tab 间共享筛选条件，特有条件独立生效
- ✅ **类型安全**: 使用枚举类管理搜索模式
- ✅ **持久化**: 保存搜索历史和筛选偏好

#### 2. 浮动的标签自动补全 (FloatingAutocompleteSuggestions)

```kotlin
@Composable
fun FloatingAutocompleteSuggestions(
    searchText: String,
    suggestions: List<String>,
    onSuggestionClick: (String) -> Unit,
    modifier: Modifier = Modifier
) {
    // 仅在满足条件时显示
    val shouldShow = remember(searchText) {
        // 1. 搜索框有文字
        // 2. 文字末尾不是空格
        // 3. 建议列表非空
        searchText.isNotBlank() && 
        !searchText.endsWith(" ") && 
        suggestions.isNotEmpty()
    }
    
    if (shouldShow) {
        Card(
            modifier = modifier
                .fillMaxWidth()
                .heightIn(max = 200.dp)
                .padding(horizontal = 16.dp),
            elevation = CardDefaults.cardElevation(8.dp)
        ) {
            LazyColumn {
                items(suggestions) { suggestion ->
                    SuggestionItem(
                        text = suggestion,
                        onClick = { onSuggestionClick(suggestion) }
                    )
                }
            }
        }
    }
}

@Composable
private fun SuggestionItem(
    text: String,
    onClick: () -> Unit
) {
    ListItem(
        headlineContent = { Text(text) },
        leadingContent = {
            Icon(Icons.Default.Tag, contentDescription = null)
        },
        modifier = Modifier.clickable(onClick = onClick)
    )
}
```

**触发逻辑**：
1. 检测输入框文字变化
2. 提取最后一个词（按空格分割）
3. 调用 `TagApi.searchTagAutocomplete(lastWord)`
4. 显示建议列表

**点击行为**：
1. 获取输入框文字按空格分割的数组
2. 替换最后一个元素为点击的标签
3. 用空格重新拼接
4. 更新输入框文字
5. 关闭建议列表

```kotlin
// ViewModel 中的逻辑
fun onSearchTextChange(newText: String) {
    _searchText.value = newText
    
    // 检查是否需要显示自动补全
    if (newText.isNotBlank() && !newText.endsWith(" ")) {
        val words = newText.split(" ")
        val lastWord = words.lastOrNull()?.trim()
        
        if (!lastWord.isNullOrBlank() && lastWord.length >= 2) {
            // 防抖后调用
            viewModelScope.launch {
                delay(300) // 防抖
                if (_searchText.value == newText) {
                    fetchTagAutocomplete(lastWord)
                }
            }
        }
    } else {
        _autocompleteSuggestions.value = emptyList()
    }
}

fun onSuggestionClick(suggestion: String) {
    val currentText = _searchText.value
    val words = currentText.split(" ").toMutableList()
    
    if (words.isNotEmpty()) {
        words[words.lastIndex] = suggestion
        _searchText.value = words.joinToString(" ") + " "
        _autocompleteSuggestions.value = emptyList()
    }
}

private suspend fun fetchTagAutocomplete(keyword: String) {
    try {
        val result = tagApi.searchTagAutocomplete(keyword)
        _autocompleteSuggestions.value = result.candidates.map { it.tag_name }
    } catch (e: Exception) {
        // 忽略错误，不影响主流程
    }
}
```

```kotlin
@Composable
fun SearchRecommendationsSection(
    recommendations: SearchSuggestionBody?,
    onTagClick: (String) -> Unit,
    onThumbnailClick: (String) -> Unit,
    modifier: Modifier = Modifier
) {
    Column(
        modifier = modifier
            .fillMaxWidth()
            .fillMaxHeight() // 占据剩余空间

            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {

            style = MaterialTheme.typography.titleMedium
        )

                TagSection(
                    title = "🔥 热门标签 (插画)",
                    tags = data.illusts,
                    onTagClick = onTagClick
                )
            }
            
            // 热门标签 - 小说
            if (data.novels.isNotEmpty()) {
                TagSection(
                    title = "🔥 热门标签 (小说)",
                    tags = data.novels,
                    onTagClick = onTagClick
                )
            }
            
            // 推荐标签
            if (data.recommendTags.tags.isNotEmpty()) {
                TagSection(
                    title = "⭐ 推荐标签",
                    tags = data.recommendTags.tags.map { it.tag },
                    onTagClick = onTagClick
                )
            }
            
            // 我的收藏标签
            if (data.recommendByTags.isNotEmpty()) {
                TagSection(
                    title = "📚 我的收藏标签",
                    tags = data.recommendByTags.map { it.tag_name },
                    onTagClick = onTagClick
                )
            }
            
            // 缩略图预览
            if (data.thumbnails.illust.isNotEmpty() || 
                data.thumbnails.novel.isNotEmpty()) {
                ThumbnailPreviewSection(
                    thumbnails = data.thumbnails,
                    onThumbnailClick = onThumbnailClick
                )
            }
        } ?: run {
            // 加载中或空状态
            Box(
                modifier = Modifier.fillMaxSize(),
                contentAlignment = Alignment.Center
            ) {
                CircularProgressIndicator()
            }
        }
    }
}
```

```kotlin
// 根据当前搜索类型决定显示哪些筛选器
@Composable
fun FilterPanel(
    searchType: SearchType,
    illustParams: IllustSearchParams,
    novelParams: NovelSearchParams,
    userParams: UserSearchParams,
    onIllustParamsChange: (IllustSearchParams) -> Unit,
    onNovelParamsChange: (NovelSearchParams) -> Unit,
    onUserParamsChange: (UserSearchParams) -> Unit
) {
    Column {
        // 搜索模式 - 所有类型都需要
        when (searchType) {
            SearchType.ILLUST -> IllustSearchModeSelector(
                selected = illustParams.searchMode,
                onSelect = { onIllustParamsChange(illustParams.copy(searchMode = it)) }
            )
            SearchType.NOVEL -> NovelSearchModeSelector(
                selected = novelParams.searchMode,
                onSelect = { onNovelParamsChange(novelParams.copy(searchMode = it)) }
            )
            SearchType.USER -> UserSearchModeSelector(
                selected = userParams.searchMode,
                onSelect = { onUserParamsChange(userParams.copy(searchMode = it)) }
            )
        }
        
        // 排序 - 仅插画和小说
        if (searchType != SearchType.USER) {
            val order = when (searchType) {
                SearchType.ILLUST -> illustParams.order
                SearchType.NOVEL -> novelParams.order
                else -> SortOrder.DATE_DESC
            }
            SortSelector(
                selected = order,
                onSelect = {
                    when (searchType) {
                        SearchType.ILLUST -> onIllustParamsChange(illustParams.copy(order = it))
                        SearchType.NOVEL -> onNovelParamsChange(novelParams.copy(order = it))
                        else -> {}
                    }
                }
            )
        }
        
        // 内容分级 - 仅插画和小说
        if (searchType != SearchType.USER) {
            val mode = when (searchType) {
                SearchType.ILLUST -> illustParams.contentMode
                SearchType.NOVEL -> novelParams.contentMode
                else -> ContentMode.ALL
            }
            ContentModeSelector(
                selected = mode,
                onSelect = {
                    when (searchType) {
                        SearchType.ILLUST -> onIllustParamsChange(illustParams.copy(contentMode = it))
                        SearchType.NOVEL -> onNovelParamsChange(novelParams.copy(contentMode = it))
                        else -> {}
                    }
                }
            )
        }
        
        // AI过滤 - 仅插画和小说
        if (searchType != SearchType.USER) {
            val hideAi = when (searchType) {
                SearchType.ILLUST -> illustParams.hideAi
                SearchType.NOVEL -> novelParams.hideAi
                else -> false
            }
            AiFilterSwitch(
                checked = hideAi,
                onCheckedChange = {
                    when (searchType) {
                        SearchType.ILLUST -> onIllustParamsChange(illustParams.copy(hideAi = it))
                        SearchType.NOVEL -> onNovelParamsChange(novelParams.copy(hideAi = it))
                        else -> {}
                    }
                }
            )
        }
        
        // 日期范围 - 仅插画和小说
        if (searchType != SearchType.USER) {
            val dateRange = when (searchType) {
                SearchType.ILLUST -> illustParams.dateRange
                SearchType.NOVEL -> novelParams.dateRange
                else -> null
            }
            DateRangeSelector(
                dateRange = dateRange,
                onDateRangeChange = {
                    when (searchType) {
                        SearchType.ILLUST -> onIllustParamsChange(illustParams.copy(dateRange = it))
                        SearchType.NOVEL -> onNovelParamsChange(novelParams.copy(dateRange = it))
                        else -> {}
                    }
                }
            )
        }
        
        // 有作品过滤 - 仅用户
        if (searchType == SearchType.USER) {
            HasWorkSwitch(
                checked = userParams.onlyWithWork,
                onCheckedChange = {
                    onUserParamsChange(userParams.copy(onlyWithWork = it))
                }
            )
        }
    }
}
```

```kotlin
class SearchViewModel(
    private val artworkRepository: ArtworkRepository,
    private val novelRepository: NovelRepository,
    private val userRepository: UserRepository,
    private val tagApi: TagApi,
    private val searchHistoryManager: SearchHistoryManager
) : ViewModel() {
    
    private val _state = MutableStateFlow(SearchState())
    val state: StateFlow<SearchState> = _state.asStateFlow()
    
    init {
        // 加载搜索历史
        viewModelScope.launch {
            searchHistoryManager.getHistory().collect { history ->
                _state.update { it.copy(searchHistory = history) }
            }
        }
        
        // 加载搜索建议
        loadSearchRecommendations()
    }
    
    /**
     * 加载搜索建议（搜索准备页）
     */
    private fun loadSearchRecommendations() {
        viewModelScope.launch {
            _state.update { it.copy(isLoadingRecommendations = true) }
            try {
                val result = tagApi.getSearchRecommendations(mode = "all")
                if (!result.error && result.body != null) {
                    _state.update { 
                        it.copy(
                            searchRecommendations = result.body,
                            isLoadingRecommendations = false
                        )
                    }
                }
            } catch (e: Exception) {
                _state.update { 
                    it.copy(
                        isLoadingRecommendations = false,
                        error = e.message
                    )
                }
            }
        }
    }
    
    /**
     * 搜索关键词变化（同时用于准备页和结果页）
     */
    fun onSearchKeywordChange(keyword: String) {
        _state.update { it.copy(searchKeyword = keyword) }
        
        // 触发标签自动补全
        if (keyword.isNotBlank() && !keyword.endsWith(" ")) {
            val words = keyword.split(" ")
            val lastWord = words.lastOrNull()?.trim()
            
            if (!lastWord.isNullOrBlank() && lastWord.length >= 2) {
                viewModelScope.launch {
                    delay(300) // 防抖
                    if (_state.value.searchKeyword == keyword) {
                        fetchTagAutocomplete(lastWord)
                    }
                }
            }
        } else {
            _state.update { it.copy(autocompleteSuggestions = emptyList()) }
        }
    }
    
    /**
     * 获取标签自动补全（同时用于准备页和结果页）
     */
    private suspend fun fetchTagAutocomplete(keyword: String) {
        try {
            val result = tagApi.searchTagAutocomplete(keyword)
            _state.update { 
                it.copy(autocompleteSuggestions = result.candidates.map { c -> c.tag_name }) 
            }
        } catch (e: Exception) {
            // 忽略错误，不影响主流程
        }
    }
    
    /**
     * 点击自动补全建议（同时用于准备页和结果页）
     */
    fun onAutocompleteSuggestionClick(suggestion: String) {
        val currentKeyword = _state.value.searchKeyword
        val words = currentKeyword.split(" ").toMutableList()
        
        if (words.isNotEmpty()) {
            words[words.lastIndex] = suggestion
            val newKeyword = words.joinToString(" ") + " "
            _state.update { 
                it.copy(
                    searchKeyword = newKeyword,
                    autocompleteSuggestions = emptyList()
                )
            }
        }
    }
    
    /**
     * 点击搜索历史
     */
    fun onHistoryClick(keyword: String) {
        _state.update { it.copy(searchKeyword = keyword) }
        performSearch()
    }
    
    /**
     * 清空搜索历史
     */
    fun clearHistory() {
        viewModelScope.launch {
            searchHistoryManager.clearHistory()
        }
    }
    
    /**
     * 执行搜索
     */
    fun performSearch() {
        val keyword = _state.value.searchKeyword.trim()
        if (keyword.isBlank()) return
        
        // 保存到搜索历史
        viewModelScope.launch {
            searchHistoryManager.addHistory(keyword)
        }
        
        // 切换到搜索结果页
        _state.update { 
            it.copy(
                currentPage = SearchPage.RESULTS,
                illustPage = 1,
                novelPage = 1,
                userPage = 1,
                illustResults = emptyList(),
                novelResults = emptyList(),
                userResults = emptyList()
            )
        }
        
        // 执行当前分类的搜索
        searchCurrentCategory()
    }
    
    /**
     * 搜索当前分类
     */
    private fun searchCurrentCategory() {
        when (_state.value.currentCategory) {
            SearchCategory.ILLUST -> searchIllusts()
            SearchCategory.NOVEL -> searchNovels()
            SearchCategory.USER -> searchUsers()
        }
    }
    
    /**
     * 搜索插画
     */
    private fun searchIllusts() {
        val state = _state.value
        if (!state.hasMoreIllust || state.isSearching) return
        
        viewModelScope.launch {
            _state.update { it.copy(isSearching = true) }
            
            val result = artworkRepository.searchArtworks(
                keyword = state.searchKeyword,
                page = state.illustPage,
                searchMode = state.illustParams.searchMode.value,
                order = state.illustParams.order.value,
                aiType = if (state.illustParams.hideAi) 1 else null
            )
            
            result.fold(
                onSuccess = { artworks ->
                    _state.update { current ->
                        current.copy(
                            illustResults = current.illustResults + artworks,
                            illustPage = current.illustPage + 1,
                            hasMoreIllust = artworks.isNotEmpty(),
                            isSearching = false,
                            error = null
                        )
                    }
                },
                onFailure = { error ->
                    _state.update { 
                        it.copy(
                            isSearching = false,
                            error = error.message
                        )
                    }
                }
            )
        }
    }
    
    /**
     * 搜索小说
     */
    private fun searchNovels() {
        val state = _state.value
        if (!state.hasMoreNovel || state.isSearching) return
        
        viewModelScope.launch {
            _state.update { it.copy(isSearching = true) }
            
            val result = novelRepository.searchNovels(
                keyword = state.searchKeyword,
                page = state.novelPage,
                searchMode = state.novelParams.searchMode.value,
                order = state.novelParams.order.value,
                aiType = if (state.novelParams.hideAi) 1 else null
            )
            
            result.fold(
                onSuccess = { novels ->
                    _state.update { current ->
                        current.copy(
                            novelResults = current.novelResults + novels,
                            novelPage = current.novelPage + 1,
                            hasMoreNovel = novels.isNotEmpty(),
                            isSearching = false,
                            error = null
                        )
                    }
                },
                onFailure = { error ->
                    _state.update { 
                        it.copy(
                            isSearching = false,
                            error = error.message
                        )
                    }
                }
            )
        }
    }
    
    /**
     * 搜索用户
     */
    private fun searchUsers() {

    }
    
    /**

     */
    fun onCategoryChange(category: SearchCategory) {
        _state.update { it.copy(currentCategory = category) }
        
        // 如果该分类还没有数据，则加载
        when (category) {
            SearchCategory.ILLUST -> {
                if (_state.value.illustResults.isEmpty()) {
```

### 缓存策略

```kotlin
// 搜索结果缓存管理器
class SearchResultCache {
    private val cache = LruCache<String, CacheEntry>(maxSize = 20)
    private val cacheTimeout = 5.minutes
    
    data class CacheEntry(
        val results: List<Any>,
        val timestamp: Instant
    )
    
    fun get(key: String): List<Any>? {
        val entry = cache.get(key) ?: return null
        
        // 检查是否过期
        if (Clock.System.now() - entry.timestamp > cacheTimeout) {
            cache.remove(key)
            return null
        }
        
        return entry.results
    }
    
    fun put(key: String, results: List<Any>) {
        cache.put(key, CacheEntry(results, Clock.System.now()))
    }
    
    fun clear() {
        cache.evictAll()
    }
}

// 使用缓存
private suspend fun searchIllusts() {
    val cacheKey = buildCacheKey(
        keyword = state.value.searchKeyword,
        params = state.value.illustParams,
        page = state.value.illustPage
    )
    
    // 先检查缓存
    searchResultCache.get(cacheKey)?.let { cached ->
        _state.update { it.copy(illustResults = cached as List<Artwork>) }
        return
    }
    
    // 缓存未命中，请求 API
    val result = artworkRepository.searchArtworks(...)
    result.onSuccess { artworks ->
        searchResultCache.put(cacheKey, artworks)
        _state.update { it.copy(illustResults = artworks) }
    }
}

private fun buildCacheKey(
    keyword: String,
    params: IllustSearchParams,
    page: Int
): String {
    return "illust_${keyword}_${params.searchMode.value}_${params.order.value}_${page}"
}
```

### 自动补全机制

**统一触发条件**（准备页 & 结果页）：
- 输入框有文字
- 文字不以空格结尾
- 最后一个词长度 ≥ 2 字符
- 300ms 防抖延迟

**统一行为**：
- 调用 `TagApi.searchTagAutocomplete(lastWord)`
- 显示浮动建议列表
- 点击建议替换最后一个词并添加空格
- 关闭建议列表
