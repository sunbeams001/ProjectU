# 登录凭据管理系统 - 快速参考

## 📦 核心组件速查

### AuthRepository
```kotlin
// 注入
val authRepository: AuthRepository = koinInject()

// 保存凭据
authRepository.saveCredentials("12345678_xxxxxxxxxxxx")

// 检查登录状态
val isLoggedIn = authRepository.isLoggedIn()

// 观察登录状态
authRepository.observeLoginState().collect { isLoggedIn ->
    // 处理登录状态变化
}

// 清除凭据（登出）
authRepository.clearCredentials()

// 获取当前用户ID
val userId = authRepository.getCurrentUserId()
```

### PixivConfigStore
```kotlin
// 注入
val configStore: PixivConfigStore = koinInject()

// 观察配置变化
configStore.config.collect { config ->
    println("PHPSESSID: ${config.phpSessionId}")
    println("Language: ${config.language}")
}

// 获取当前配置
val config = configStore.getCurrentConfig()

// 设置 PHPSESSID
configStore.setPhpSessionId("12345678_xxxxxxxxxxxx")

// 清除配置
configStore.clear()
```

## 🧪 测试代码示例

### 测试登录流程
```kotlin
@Test
fun testLoginFlow() = runBlocking {
    val authRepository: AuthRepository = get()
    
    // 初始状态：未登录
    assertFalse(authRepository.isLoggedIn())
    
    // 保存凭据
    val result = authRepository.saveCredentials("12345678_test")
    assertTrue(result.isSuccess)
    
    // 检查登录状态
    assertTrue(authRepository.isLoggedIn())
    
    // 获取用户ID
    val userId = authRepository.getCurrentUserId()
    assertEquals(12345678L, userId)
    
    // 登出
    authRepository.clearCredentials()
    assertFalse(authRepository.isLoggedIn())
}
```

### 测试 API 调用
```kotlin
@Test
fun testPixivApiWithAuth() = runBlocking {
    val authRepository: AuthRepository = get()
    val pixivApi: PixivApi = get()
    
    // 先登录
    authRepository.saveCredentials(System.getenv("PIXIV_PHPSESSID"))
    
    // 调用 API
    val response = pixivApi.illustApi.getIllustDetail("102814610")
    
    assertFalse(response.error)
    assertEquals("102814610", response.body.id)
}
```

## 📝 常用操作

### 获取当前登录用户信息
```kotlin
suspend fun getCurrentUserInfo(): User? {
    val authRepository: AuthRepository = get()
    val userId = authRepository.getCurrentUserId() ?: return null
    
    val userRepository: UserRepository = get()
    return userRepository.getUserById(userId.toString())
        .getOrNull()
}
```

### 检查登录并跳转
```kotlin
@Composable
fun ProtectedScreen() {
    val authRepository: AuthRepository = koinInject()
    val navigator = LocalNavigator.currentOrThrow
    
    LaunchedEffect(Unit) {
        if (!authRepository.isLoggedIn()) {
            navigator.replaceAll(LoginScreen())
        }
    }
    
    // 受保护的内容...
}
```

### 全局监听登录状态
```kotlin
@Composable
fun App() {
    val authRepository: AuthRepository = koinInject()
    val isLoggedIn by authRepository.observeLoginState()
        .collectAsState(initial = false)
    
    LaunchedEffect(isLoggedIn) {
        if (!isLoggedIn) {
            // 跳转到登录页面
        }
    }
    
    // ...
}
```

private fun isValidPhpSessionId(phpSessionId: String): Boolean {
    // 自定义验证逻辑
    return phpSessionId.matches(Regex("""^\d+_[a-zA-Z0-9]+$"""))
